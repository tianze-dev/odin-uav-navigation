cmake_minimum_required(VERSION 3.22)
project(odin_simulation_gz VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz_cmake_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-cmake3 REQUIRED)
gz_find_package(gz-plugin2 REQUIRED COMPONENTS register)
gz_find_package(gz-sim8 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
include(cmake/RGLDependencies.cmake)
set(upstream ${CMAKE_CURRENT_SOURCE_DIR}/third_party/rgl_gazebo)
add_library(OdinRGLSceneManager SHARED ${upstream}/src/RGLServerPluginManager.cc ${upstream}/src/Mesh.cc ${upstream}/src/Scene.cc ${upstream}/src/Utils.cc)
add_library(OdinRGLDepthSensor SHARED src/DepthSensor.cc ${upstream}/src/Utils.cc)
foreach(target OdinRGLSceneManager OdinRGLDepthSensor)
  target_include_directories(${target} PRIVATE ${upstream}/include include)
  target_link_libraries(${target} PRIVATE gz-plugin2::gz-plugin2 gz-sim8::gz-sim8 RobotecGPULidar)
  set_target_properties(${target} PROPERTIES INSTALL_RPATH "$ORIGIN")
endforeach()
add_executable(odin_ros_adapter src/RosAdapter.cc)
target_include_directories(odin_ros_adapter PRIVATE include)
target_link_libraries(odin_ros_adapter gz-sim8::gz-sim8)
ament_target_dependencies(odin_ros_adapter rclcpp sensor_msgs)
install(TARGETS OdinRGLSceneManager OdinRGLDepthSensor LIBRARY DESTINATION lib/${PROJECT_NAME}/rgl)
install(IMPORTED_RUNTIME_ARTIFACTS RobotecGPULidar LIBRARY DESTINATION lib/${PROJECT_NAME}/rgl)
install(TARGETS odin_ros_adapter RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config urdf meshes DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(FILES ${upstream}/LICENSE ${upstream}/UPSTREAM.md
  DESTINATION share/${PROJECT_NAME}/third_party/rgl_gazebo)
ament_environment_hooks(hooks/odin_gz.dsv.in)
ament_package()
