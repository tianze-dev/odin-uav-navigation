cmake_minimum_required(VERSION 3.16)
project(uav_rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
foreach(dep rclcpp rviz_common rviz_rendering pluginlib geometry_msgs nav_msgs std_msgs std_srvs uav_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(uav_navigation_panel SHARED src/navigation_panel.cpp src/goal_plane_tool.cpp include/uav_rviz_plugins/navigation_panel.hpp)
target_include_directories(uav_navigation_panel PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(uav_navigation_panel rclcpp rviz_common rviz_rendering pluginlib geometry_msgs nav_msgs std_msgs std_srvs uav_interfaces)
target_link_libraries(uav_navigation_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS uav_navigation_panel LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/navigation_visuals DESTINATION lib/${PROJECT_NAME})

ament_package()
