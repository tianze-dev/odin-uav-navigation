cmake_minimum_required(VERSION 3.16)
project(path_searching)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
add_library(path_searching SHARED src/dyn_a_star.cpp)
target_include_directories(path_searching PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${EIGEN3_INCLUDE_DIR})
ament_target_dependencies(path_searching rclcpp plan_env)
install(TARGETS path_searching EXPORT export_path_searching ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_path_searching HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_libraries(path_searching)
ament_export_dependencies(rclcpp plan_env Eigen3)
ament_package()
