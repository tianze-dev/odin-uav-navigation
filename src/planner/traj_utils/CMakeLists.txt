cmake_minimum_required(VERSION 3.16)
project(traj_utils)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/PolyTraj.msg" DEPENDENCIES std_msgs geometry_msgs builtin_interfaces)
add_library(traj_utils_lib SHARED src/planning_visualization.cpp)
target_include_directories(traj_utils_lib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${EIGEN3_INCLUDE_DIR})
ament_target_dependencies(traj_utils_lib rclcpp std_msgs geometry_msgs visualization_msgs builtin_interfaces)
install(TARGETS traj_utils_lib EXPORT export_traj_utils ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_traj_utils HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_libraries(traj_utils_lib)
ament_export_dependencies(rclcpp std_msgs geometry_msgs visualization_msgs builtin_interfaces Eigen3 rosidl_default_runtime)

ament_package()
