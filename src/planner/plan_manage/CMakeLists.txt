cmake_minimum_required(VERSION 3.16)
project(ego_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep rclcpp nav_msgs geometry_msgs std_msgs std_srvs uav_interfaces quadrotor_msgs plan_env path_searching traj_opt traj_utils visualization_msgs)
 find_package(${dep} REQUIRED)
endforeach()
add_executable(ego_planner_node src/ego_planner_node.cpp src/ego_replan_fsm.cpp src/planner_manager.cpp)
target_include_directories(ego_planner_node PRIVATE include ${EIGEN3_INCLUDE_DIR})
ament_target_dependencies(ego_planner_node rclcpp nav_msgs geometry_msgs std_msgs std_srvs uav_interfaces quadrotor_msgs plan_env path_searching traj_opt traj_utils visualization_msgs)
install(TARGETS ego_planner_node DESTINATION lib/${PROJECT_NAME})

ament_package()
