cmake_minimum_required(VERSION 3.16)
project(traj_opt)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(plan_env REQUIRED)
find_package(path_searching REQUIRED)
find_package(traj_utils REQUIRED)
add_library(traj_opt SHARED src/poly_traj_optimizer.cpp)
target_include_directories(traj_opt PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${EIGEN3_INCLUDE_DIR})
ament_target_dependencies(traj_opt rclcpp plan_env path_searching traj_utils)
install(TARGETS traj_opt EXPORT export_traj_opt ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_traj_opt HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_libraries(traj_opt)
ament_export_dependencies(rclcpp plan_env path_searching traj_utils Eigen3)
ament_package()
